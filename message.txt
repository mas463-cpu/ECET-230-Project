//  AIR MOUSE — ATmega32U4 Custom PCB
//  Final build firmware
//  Board: Arduino Leonardo (ATmega32U4)
// ============================================================

#include <Wire.h>
#include <Mouse.h>
#include <I2Cdev.h>
#include <MPU6050.h>
#include <math.h>

// ======================== PIN MAP ============================
// Match these to your schematic — change here if wiring differs
#define BTN_LEFT   A1    // SW1  (PF6) — left click
#define BTN_RIGHT  A2    // SW2  (PF5) — right click
#define BTN_RECAL  A3    // SW3  (PF4) — recalibrate gyro

#define LED_GREEN  6     // D3 status LED (PD7)
#define LED_YELLOW 4     // D4 status LED (PD4)
// D5 power LED is hardwired on 3.3V rail — always on

// ==================== TUNING KNOBS ==========================
// Adjust these to change how the cursor feels

// Sensitivity — higher = slower cursor. Start at 300, go down for faster.
#define YAW_DIV    280.0f    // left/right sensitivity
#define ROLL_DIV   280.0f    // up/down sensitivity

// Deadzone — kills jitter when the device sits still. 1–3 is typical.
#define DEADZONE_X 1
#define DEADZONE_Y 1

// Smoothing — 0.0 = max smooth (laggy), 1.0 = raw (jittery). 0.35 is a good start.
#define SMOOTH_X   0.35f
#define SMOOTH_Y   0.35f

// Adaptive bias — how quickly the resting offset drifts. Don't touch unless you know why.
#define STILL_THRESH 80.0f
#define BIAS_FAST    0.005f
#define BIAS_SLOW    (1.0f - BIAS_FAST)

// How many samples to average during calibration
#define CAL_SAMPLES  500

// ===================== GLOBALS ===============================
MPU6050 mpu;

int16_t ax, ay, az, gx, gy, gz;    // raw sensor readings
float   gx_off = 0, gz_off = 0;    // startup calibration offsets
float   gx_bias = 0, gz_bias = 0;  // running adaptive bias

bool lastLeft  = false;
bool lastRight = false;
bool lastRecal = false;

// ===================== HELPERS ===============================

int applyDeadzone(int val, int zone) {
  if (val > zone)  return val - zone;
  if (val < -zone) return val + zone;
  return 0;
}

void calibrateGyro() {
  // Blink yellow LED while calibrating — keep device STILL
  digitalWrite(LED_YELLOW, HIGH);

  long sumGx = 0, sumGz = 0;

  for (int i = 0; i < CAL_SAMPLES; i++) {
    mpu.getMotion6(&ax, &ay, &az, &gx, &gy, &gz);
    sumGx += gx;
    sumGz += gz;
    delay(3);
  }

  gx_off = (float)sumGx / CAL_SAMPLES;
  gz_off = (float)sumGz / CAL_SAMPLES;

  // Reset running bias to zero after fresh calibration
  gx_bias = 0;
  gz_bias = 0;

  digitalWrite(LED_YELLOW, LOW);

  Serial.print("Calibration done — gx_off: ");
  Serial.print(gx_off);
  Serial.print("  gz_off: ");
  Serial.println(gz_off);
}

void handleButtons() {
  bool leftNow  = (digitalRead(BTN_LEFT)  == LOW);
  bool rightNow = (digitalRead(BTN_RIGHT) == LOW);
  bool recalNow = (digitalRead(BTN_RECAL) == LOW);

  // Left click — press/release on edges only
  if (leftNow && !lastLeft)    Mouse.press(MOUSE_LEFT);
  if (!leftNow && lastLeft)    Mouse.release(MOUSE_LEFT);

  // Right click
  if (rightNow && !lastRight)  Mouse.press(MOUSE_RIGHT);
  if (!rightNow && lastRight)  Mouse.release(MOUSE_RIGHT);

  // Recalibrate — trigger on press edge
  if (recalNow && !lastRecal) {
    Serial.println("Recalibrating...");
    // Release any held mouse buttons first
    Mouse.release(MOUSE_LEFT);
    Mouse.release(MOUSE_RIGHT);
    calibrateGyro();
  }

  lastLeft  = leftNow;
  lastRight = rightNow;
  lastRecal = recalNow;
}

// ======================== SETUP ==============================
void setup() {
  Serial.begin(115200);

  // LED pins
  pinMode(LED_GREEN,  OUTPUT);
  pinMode(LED_YELLOW, OUTPUT);
  digitalWrite(LED_GREEN,  LOW);
  digitalWrite(LED_YELLOW, LOW);

  // Button pins — internal pull-ups, buttons short to GND
  pinMode(BTN_LEFT,  INPUT_PULLUP);
  pinMode(BTN_RIGHT, INPUT_PULLUP);
  pinMode(BTN_RECAL, INPUT_PULLUP);

  // I2C bus
  Wire.begin();
  Wire.setClock(100000);              // 100 kHz standard mode
  Wire.setWireTimeout(3000, true);    // 3 ms timeout, auto-reset

  // Initialize MPU-6050
  mpu.initialize();

  // Verify sensor is alive
  Serial.print("WHO_AM_I: 0x");
  Serial.println(mpu.getDeviceID(), HEX);

  if (mpu.testConnection()) {
    Serial.println("MPU-6050 connected.");
    digitalWrite(LED_GREEN, HIGH);    // green ON = sensor OK
  } else {
    Serial.println("MPU-6050 NOT found!");
    // Blink green rapidly to signal error
    while (true) {
      digitalWrite(LED_GREEN, HIGH); delay(100);
      digitalWrite(LED_GREEN, LOW);  delay(100);
    }
  }

  // Calibrate gyroscope — keep device still!
  Serial.println("Calibrating — hold still...");
  calibrateGyro();

  // Start USB HID mouse
  Mouse.begin();
  Serial.println("Mouse active.");
}

// ========================= LOOP ==============================
void loop() {
  static float dx_f = 0.0f;
  static float dy_f = 0.0f;

  // ---- Handle button presses first ----
  handleButtons();

  // ---- Read all 6 axes from MPU-6050 ----
  mpu.getMotion6(&ax, &ay, &az, &gx, &gy, &gz);

  // ---- Subtract startup calibration offset ----
  float yaw_rate  = (float)(gz - gz_off);   // left/right
  float roll_rate = (float)(gx - gx_off);   // up/down

  // ---- Adaptive bias correction (kills slow drift) ----
  // When mostly still, slowly pull the bias toward current reading
  if (fabs(roll_rate) < STILL_THRESH) {
    gx_bias = BIAS_SLOW * gx_bias + BIAS_FAST * roll_rate;
  }
  if (fabs(yaw_rate) < STILL_THRESH) {
    gz_bias = BIAS_SLOW * gz_bias + BIAS_FAST * yaw_rate;
  }

  roll_rate -= gx_bias;
  yaw_rate  -= gz_bias;

  // ---- Scale to pixel movement ----
  int dx = (int)(yaw_rate  / YAW_DIV);
  int dy = (int)(roll_rate / ROLL_DIV);

  // Flip X axis if cursor moves opposite to your wrist
  dx = -dx;

  // ---- Deadzone — zero out micro-tremors ----
  dx = applyDeadzone(dx, DEADZONE_X);
  dy = applyDeadzone(dy, DEADZONE_Y);

  // ---- Exponential smoothing — reduces jitter ----
  dx_f = (1.0f - SMOOTH_X) * dx_f + SMOOTH_X * dx;
  dy_f = (1.0f - SMOOTH_Y) * dy_f + SMOOTH_Y * dy;

  // ---- Send to host as USB mouse movement ----
  Mouse.move((int)dx_f, (int)(-dy_f));
}
